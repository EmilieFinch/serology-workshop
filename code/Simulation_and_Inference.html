Code 
final_year    <- 8
n_individuals <- 3

outcome_matrix <- matrix(NA, 
                         nrow = n_individuals,
                         ncol = final_year)

FOI                 <- 0.5
prob_inf            <- rep(1 - exp(-FOI), n_individuals)
previously_infected <- rep(0, n_individuals)

set.seed(2236)

for(year_of_age in 1:final_year)
{
  prob_inf <- ifelse(previously_infected, 0, prob_inf)
  
  outcomes <- rbern(n = n_individuals, prob = prob_inf)
  
  outcome_matrix[, year_of_age] <- outcomes
  
  previously_infected <- previously_infected | outcomes
}

log_likelihood_function <- function(FOI)
{
  if(FOI < 0) return -Inf
  
  y_a     <- seropositive_proportion_by_age_df$n_seropositive
  
  n_a     <- seropositive_proportion_by_age_df$n_individuals
  
  ages    <- seropositive_proportion_by_age_df$age
  
  prob_a  <-  1 - exp(-FOI * ages) # Serocatalytic model
  
  log_likelihood_values <- dbinom(x    = y_a,
                                  size = n_a,
                                  prob = prob_a,
                                  log  = TRUE)
  
  -sum(log_likelihood_values)
}
